cmake_minimum_required(VERSION 3.16)
project(tuning_scoped_probe)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rclcpp_lifecycle REQUIRED)
find_package(std_srvs REQUIRED)
find_package(nav2_regulated_pure_pursuit_controller REQUIRED)
find_package(nav2_core REQUIRED)
find_package(nav2_behavior_tree REQUIRED)
find_package(nav2_bt_navigator REQUIRED)
find_package(patrolbot_interfaces REQUIRED)
add_executable(cache_probe ../tuning_scoped_cache_probe.cpp)
target_compile_features(cache_probe PUBLIC cxx_std_17)
ament_target_dependencies(cache_probe rclcpp rclcpp_lifecycle std_srvs
  nav2_regulated_pure_pursuit_controller)
add_executable(muxer_probe ../tuning_scoped_muxer_probe.cpp)
target_compile_features(muxer_probe PUBLIC cxx_std_17)
target_compile_options(muxer_probe PRIVATE -UNDEBUG)
ament_target_dependencies(muxer_probe nav2_core nav2_behavior_tree)
add_executable(action_input_probe ../tuning_action_input_probe.cpp)
target_compile_features(action_input_probe PUBLIC cxx_std_17)
target_compile_options(action_input_probe PRIVATE -UNDEBUG)
ament_target_dependencies(action_input_probe nav2_bt_navigator nav2_util nav2_msgs
  patrolbot_interfaces rclcpp rclcpp_action std_msgs)
